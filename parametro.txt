El usuario funcional me paso la siguiente query con la que el construyo la definición. Query que el me dice que la use para construir la etl porque esa realizo toda la construcción correctamente:


DROP TABLE IF EXISTS proceso_vspc_finanzas.tmp_ceautoedt_base PURGE
;

CREATE TABLE proceso_vspc_finanzas.tmp_ceautoedt_base STORED AS PARQUET AS
SELECT
  CAST(concepto_contable AS STRING) AS concepto_contable,
  CAST(codigo_contable AS STRING) AS codigo_contable,
  CAST(SUM(monto_trx) AS DECIMAL(38, 2)) AS monto_trx
FROM
  proceso_vspc_finanzas.visionr_ceautoedt
WHERE
  ingestion_year = YEAR(NOW())
  AND ingestion_month = MONTH(NOW())
GROUP BY
  CAST(concepto_contable AS STRING),
  CAST(codigo_contable AS STRING)
;

COMPUTE INCREMENTAL STATS proceso_vspc_finanzas.tmp_ceautoedt_base
;

DROP TABLE IF EXISTS proceso_vspc_finanzas.tmp_allinf_contable PURGE
;

CREATE TABLE proceso_vspc_finanzas.tmp_allinf_contable STORED AS PARQUET AS WITH ultima_version AS (
  SELECT
    YEAR,
    ingestion_month,
    ingestion_day
  FROM
    s_apoyo_financiero.ccsap_visionr_allinf
  WHERE
    YEAR IN (YEAR(NOW()), YEAR(NOW()) - 1)
  GROUP BY
    YEAR,
    ingestion_month,
    ingestion_day
  ORDER BY
    YEAR DESC,
    ingestion_month DESC,
    ingestion_day DESC
  LIMIT
    1
)
SELECT
  CAST(a.hatpgl AS STRING) AS codigo_contable,
  CAST(a.hatprc AS STRING) AS concepto_contable,
  CAST(a.hanoda AS STRING) AS bvl_cta_deb,
  CAST(a.hanoca AS STRING) AS bvl_cta_cre
FROM
  s_apoyo_financiero.ccsap_visionr_allinf a
  INNER JOIN ultima_version uv ON a.year = uv.year
  AND a.ingestion_month = uv.ingestion_month
  AND a.ingestion_day = uv.ingestion_day
WHERE
  a.year IN (YEAR(NOW()), YEAR(NOW()) - 1)
;

COMPUTE INCREMENTAL STATS proceso_vspc_finanzas.tmp_allinf_contable
;

DROP TABLE IF EXISTS proceso_vspc_finanzas.tmp_peq_bvl_sap PURGE
;

CREATE TABLE proceso_vspc_finanzas.tmp_peq_bvl_sap STORED AS PARQUET AS
SELECT
  CAST(valor_dato_origen AS STRING) AS valor_dato_origen,
  CAST(valor_dato_destino AS STRING) AS valor_dato_destino
FROM
  s_productos.peqc_pequiv_vw_cache_peq
WHERE
  YEAR = YEAR(NOW())
  AND ingestion_month = MONTH(NOW())
  AND ingestion_day = 1
  AND codigo_aplicacion_origen = 'BVG'
  AND codigo_aplicacion_destino = 'SAP-FI'
GROUP BY
  CAST(valor_dato_origen AS STRING),
  CAST(valor_dato_destino AS STRING)
;

COMPUTE INCREMENTAL STATS proceso_vspc_finanzas.tmp_peq_bvl_sap
;

DROP TABLE IF EXISTS proceso_vspc_finanzas.tmp_consolidado_homologacion_sap PURGE
;

CREATE TABLE proceso_vspc_finanzas.tmp_consolidado_homologacion_sap STORED AS PARQUET AS
SELECT
  CAST(base.concepto_contable AS STRING) AS concepto_contable,
  CAST(base.codigo_contable AS STRING) AS codigo_contable,
  CAST(base.monto_trx AS DECIMAL(38, 2)) AS monto_trx,
  CAST(allinf.bvl_cta_deb AS STRING) AS bvl_cta_deb,
  CAST(peq_deb.valor_dato_destino AS STRING) AS sap_cta_deb,
  CAST(allinf.bvl_cta_cre AS STRING) AS bvl_cta_cre,
  CAST(peq_cre.valor_dato_destino AS STRING) AS sap_cta_cre,
  DAY(CURRENT_DATE()) AS DAY,
  MONTH(CURRENT_DATE()) AS MONTH,
  YEAR(CURRENT_DATE()) AS YEAR
FROM
  proceso_vspc_finanzas.tmp_ceautoedt_base base
  LEFT JOIN proceso_vspc_finanzas.tmp_allinf_contable allinf ON base.codigo_contable = allinf.codigo_contable
  AND base.concepto_contable = allinf.concepto_contable
  LEFT JOIN proceso_vspc_finanzas.tmp_peq_bvl_sap peq_deb ON allinf.bvl_cta_deb = peq_deb.valor_dato_origen
  LEFT JOIN proceso_vspc_finanzas.tmp_peq_bvl_sap peq_cre ON allinf.bvl_cta_cre = peq_cre.valor_dato_origen
;

COMPUTE INCREMENTAL STATS proceso_vspc_finanzas.tmp_consolidado_homologacion_sap
;
